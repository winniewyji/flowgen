#!/usr/bin/env python3
"""FlowGen Txt Pack Restore Script - 用法: python restore.py"""
import os, sys, tarfile, shutil

def restore(tar_path='flowgen_txt.tar.gz'):
    print(f'📦 解压: {tar_path}')
    with tarfile.open(tar_path, 'r:gz') as tar:
        tar.extractall('.')
    manifest_path = 'txt_pack/manifest.txt'
    output_dir = 'restored'
    os.makedirs(output_dir, exist_ok=True)
    total = success = 0
    with open(manifest_path) as f:
        for line in f:
            if '|' not in line: continue
            total += 1
            txt_name, orig_path = line.strip().split('|')
            src = f'txt_pack/{txt_name}'
            if os.path.exists(src):
                d = os.path.dirname(orig_path)
                if d: os.makedirs(os.path.join(output_dir, d), exist_ok=True)
                shutil.copy2(src, os.path.join(output_dir, orig_path))
                success += 1
    shutil.rmtree('txt_pack')
    print(f'✅ 还原完成！{success}/{total} 个文件 -> {output_dir}/')

if __name__ == '__main__':
    tar_path = sys.argv[1] if len(sys.argv) > 1 else 'flowgen_txt.tar.gz'
    if os.path.exists(tar_path): restore(tar_path)
    else: print(f'❌ 文件不存在: {tar_path}')
